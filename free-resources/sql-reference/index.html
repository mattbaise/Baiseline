SELECT

Read & Filter

SELECT id, name, status
FROM shipments
WHERE status = 'DELIVERED'
  AND weight_kg > 100
ORDER BY name
LIMIT 25;
AGGREGATE

Group & Measure

SELECT carrier_id,
       COUNT(*) AS shipments,
       AVG(weight_kg) AS avg_weight
FROM shipments
GROUP BY carrier_id
HAVING COUNT(*) >= 5
ORDER BY shipments DESC;
JOINS

Combine Tables

SELECT s.id, c.carrier_name
FROM shipments AS s
JOIN carriers AS c
  ON c.carrier_id = s.carrier_id;

-- Keep unmatched left rows
SELECT * FROM a
LEFT JOIN b ON b.a_id = a.id;
WRITE

Insert, Update, Delete

INSERT INTO carriers (name, active)
VALUES ('Acme Freight', TRUE);

UPDATE carriers
SET active = FALSE
WHERE id = 7;

DELETE FROM temp_rows
WHERE created_at < CURRENT_DATE - INTERVAL '30 days';
DDL

Create Tables

CREATE TABLE carriers (
  carrier_id BIGSERIAL PRIMARY KEY,
  carrier_name TEXT NOT NULL,
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CONSTRAINTS

Protect Data

email TEXT UNIQUE NOT NULL,
weight_kg NUMERIC CHECK (weight_kg > 0),
carrier_id BIGINT REFERENCES carriers(carrier_id),
status TEXT CHECK (status IN (
  'PENDING','IN_TRANSIT','DELIVERED','CANCELLED'
))
NULL

NULL & CASE

SELECT COALESCE(delay_minutes, 0)
FROM shipments;

SELECT CASE
  WHEN delay_minutes > 60 THEN 'LATE'
  WHEN delay_minutes > 0 THEN 'DELAYED'
  ELSE 'ON_TIME'
END AS performance
FROM shipments;
CTE

CTEs

WITH delivered AS (
  SELECT *
  FROM shipments
  WHERE status = 'DELIVERED'
)
SELECT carrier_id, COUNT(*)
FROM delivered
GROUP BY carrier_id;
TRANSACTION

Transactions

BEGIN;

UPDATE accounts
SET balance = balance - 100
WHERE id = 1;

UPDATE accounts
SET balance = balance + 100
WHERE id = 2;

COMMIT;
-- ROLLBACK; on failure
SAFETY

SQL Habits

  • Use parameterized queries for user input.
  • Use constraints to enforce important rules.
  • Index columns used frequently for filtering/joining after measuring.
  • Preview rows before destructive updates/deletes.
  • Use transactions for related writes.
  • Back up important data before risky migrations.
